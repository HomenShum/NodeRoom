NS={'s':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
def workbook(path):
    with zipfile.ZipFile(path) as z:
        strings=[]
        if 'xl/sharedStrings.xml' in z.namelist():
            strings=[''.join(t.text or '' for t in si.findall('.//s:t',NS)) for si in ET.fromstring(z.read('xl/sharedStrings.xml')).findall('s:si',NS)]
        styles=z.read('xl/styles.xml')
        style_tree=ET.fromstring(styles);xf=style_tree.find('s:cellXfs',NS)
        cells={}
        for c in ET.fromstring(z.read('xl/worksheets/sheet1.xml')).findall('.//s:c',NS):
            v=c.find('s:v',NS);raw=None if v is None else v.text;t=c.get('t')
            if t=='s':value=strings[int(raw)] if raw is not None else None
            elif t=='inlineStr':value=''.join(i.text or '' for i in c.findall('.//s:t',NS))
            elif raw is None:value=None
            elif t=='b':value=raw=='1'
            elif t in ('str','e'):value=raw
            else:
                try:value=int(raw)
                except ValueError:
                    try:value=float(raw)
                    except ValueError:value=raw
            f=c.find('s:f',NS);s=int(c.get('s','0'))
            cells[c.attrib['r']]={'value':value,'formula':None if f is None else {'text':f.text,'attributes':f.attrib},'styleIndex':s,'styleXml':ET.tostring(xf[s],encoding='unicode')}
        for row in range(1,7):
            for col in 'ABCDE':
                key=f'{col}{row}'
                cells.setdefault(key,{'value':None,'formula':None,'styleIndex':0,'styleXml':ET.tostring(xf[0],encoding='unicode')})
        return {'file':bind(path,path.relative_to(O).as_posix()),'sheetNames':[s.get('name') for s in ET.fromstring(z.read('xl/workbook.xml')).findall('s:sheets/s:sheet',NS)],'stylesSha256':sha(styles),'cells':cells,'zipMembers':z.namelist()}
