// @vitest-environment jsdom
import ExcelJS from "exceljs";
import { createElement } from "react";
import { fireEvent, render, screen, waitFor } from "@testing-library/react";
import { describe, expect, it, vi } from "vitest";
import type { Actor, Artifact } from "../src/engine/types";
import { SHEET_ROWS } from "../src/engine/demoRoom";

const exportFixture = vi.hoisted(() => ({
  store: {} as Record<string, unknown>,
  bytes: [] as ArrayBuffer[],
}));
vi.mock("../src/app/store", () => ({ useStore: () => exportFixture.store }));
// Keep the real component projection and ExcelJS serialization. Only replace browser dispatch;
// the companion browser scenario saves and reopens the actual downloaded file.
vi.mock("../src/ui/workArtifacts/xlsxDownload", () => ({
  downloadXlsxWorkbook: async (workbook: ExcelJS.Workbook, fileName: string, rowCount: number) => {
    const bytes = await workbook.xlsx.writeBuffer();
    exportFixture.bytes.push(bytes);
    return { fileName, rowCount, sheetCount: workbook.worksheets.length, byteCount: bytes.byteLength, at: "test" };
  },
}));

import { Artifact as ArtifactPanel, applyExportCell, exportCellValue, spreadsheetCellFontColorCss } from "../src/ui/panels/Artifact";

describe("NodeRoom XLSX cell export", () => {
  it("preserves a managed formula, cached result, and number format after reopen", async () => {
    const workbook = new ExcelJS.Workbook();
    const sheet = workbook.addWorksheet("Model");

    applyExportCell(sheet.getCell("C2"), {
      value: 30,
      formula: "=B3*2",
      numFmt: "$#,##0",
    });

    const reopened = new ExcelJS.Workbook();
    await reopened.xlsx.load(await workbook.xlsx.writeBuffer());
    const cell = reopened.getWorksheet("Model")!.getCell("C2");

    expect(cell.formula).toBe("B3*2");
    expect(cell.result).toBe(30);
    expect(cell.numFmt).toBe("$#,##0");
  });

  it("continues coercing scalar numeric strings without inventing formulas", () => {
    expect(exportCellValue("3.5")).toBe(3.5);
    expect(exportCellValue({ value: "Revenue", numFmt: "General" })).toBe("Revenue");
  });

  it("overlays payload font color without losing captured font and style fields", async () => {
    const workbook = new ExcelJS.Workbook();
    const sheet = workbook.addWorksheet("Model");

    applyExportCell(
      sheet.getCell("C2"),
      { value: 30, fontColor: "80112233", numFmt: "0.0%" },
      { b: 1, i: 1, u: 1, fc: "#445566", bg: "#DDEEFF", a: "c", ind: 2, bt: 1, bb: 1, f: 0 },
      ["$#,##0"],
    );

    const reopened = new ExcelJS.Workbook();
    await reopened.xlsx.load(await workbook.xlsx.writeBuffer());
    const cell = reopened.getWorksheet("Model")!.getCell("C2");

    expect(cell.font).toMatchObject({
      bold: true,
      italic: true,
      underline: true,
      color: { argb: "80112233" },
    });
    expect(cell.fill).toMatchObject({ type: "pattern", pattern: "solid", fgColor: { argb: "FFDDEEFF" } });
    expect(cell.alignment).toMatchObject({ horizontal: "center", indent: 2 });
    expect(cell.border.top?.style).toBe("thin");
    expect(cell.border.bottom?.style).toBe("thin");
    expect(cell.numFmt).toBe("0.0%");
  });

  it("uses payload color over imported color in the grid and preserves alpha in CSS", () => {
    const artifact = {
      id: "model",
      roomId: "room",
      kind: "sheet",
      title: "Model",
      version: 1,
      updatedAt: 1,
      order: ["A1", "A2"],
      elements: {
        A1: { id: "A1", version: 1, value: { value: "Override", fontColor: "80112233" }, updatedAt: 1, updatedBy: { kind: "user", id: "u1", name: "User" } },
        A2: { id: "A2", version: 1, value: { value: "Imported" }, updatedAt: 1, updatedBy: { kind: "user", id: "u1", name: "User" } },
      },
      meta: {
        excelGrid: {
          sourceFile: "model.xlsx",
          sheetName: "Model",
          sheetNames: ["Model"],
          parser: "exceljs:xlsx-grid",
          rows: 2,
          columns: 1,
          styles: { A1: { fc: "#445566" }, A2: { fc: "#AABBCC" } },
        },
      },
    } satisfies Artifact;

    expect(spreadsheetCellFontColorCss(artifact, "A1")).toBe("#11223380");
    expect(spreadsheetCellFontColorCss(artifact, "A2")).toBe("#AABBCC");
  });

  it.each(["current sheet", "sibling sheet"] as const)("keeps Account labels through repeated reviewer exports of the %s", async (surface) => {
    const actor: Actor = { kind: "user", id: "reviewer", name: "Reviewer" };
    const canonical: Artifact = {
      id: "variance", roomId: "review-room", title: "Q3 variance", kind: "sheet",
      version: 1, updatedAt: 1, order: [], elements: {},
    };
    const rowIds = SHEET_ROWS.map((row, index) => surface === "current sheet" ? row.id : String(index + 1));
    SHEET_ROWS.forEach((row, index) => {
      const values = { label: row.label, q2: row.q2, q3: row.q3,
        variance: { value: 0.24, formula: "=C2/B2-1", numFmt: "0.0%" }, note: "Initial review" };
      for (const [column, value] of Object.entries(values)) {
        const id = `${rowIds[index]}__${column}`;
        canonical.order.push(id);
        canonical.elements[id] = { id, value, version: 1, updatedAt: 1, updatedBy: actor };
      }
    });
    const summary: Artifact = {
      id: "summary", roomId: canonical.roomId, title: "Summary", kind: "sheet",
      version: 1, updatedAt: 1, order: ["A1", "B1"],
      elements: {
        A1: { id: "A1", value: { value: "Review workbook" }, version: 1, updatedAt: 1, updatedBy: actor },
        B1: { id: "B1", value: { value: 2, formula: "=1+1", numFmt: "0.00" }, version: 1, updatedAt: 1, updatedBy: actor },
      },
    };
    if (surface === "sibling sheet") {
      // Existing multi-sheet metadata supplies numeric row IDs. The active generic sheet and
      // its canonical sibling must both survive the same workbook-export action.
      for (const art of [summary, canonical]) art.meta = {
        upload: { fileName: "review.xlsx", mimeType: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", size: 1, parsedAt: 1 },
        excelGrid: { sourceFile: "review.xlsx", sheetName: art.title, sheetNames: ["Summary", "Q3 variance"], parser: "exceljs:xlsx-grid", rows: art === canonical ? 5 : 1, columns: art === canonical ? 5 : 2 },
      };
      summary.meta!.dataframe = {
        rowCount: 1, columns: ["A", "B"].map((id, order) => ({ id, label: id, order })),
      };
    }
    const artifacts = surface === "current sheet" ? [canonical] : [summary, canonical];
    exportFixture.bytes = [];
    exportFixture.store = {
      mode: "engine", listArtifacts: () => artifacts, getArtifact: (id: string) => artifacts.find(art => art.id === id),
      listMessages: () => [], listProposals: () => [], listTraces: () => [], listDrafts: () => [],
      listMembers: () => [{ id: actor.id, actor, role: "host" }], listSessions: () => [], listPresence: () => [],
      lockFor: () => null, canUndo: () => false, updatePresence: () => undefined, clearPresence: () => undefined,
      getRoom: () => ({ id: canonical.roomId, title: "Review room" }),
      lastRun: () => null, lastLongFreeJob: () => null, lastLongFreeJobAttempts: () => [], lastLongFreeJobDetail: () => null,
    };
    const props = { roomId: canonical.roomId, me: actor, artId: artifacts[0].id, onArt: vi.fn() };
    const view = render(createElement(ArtifactPanel, props));
    if (surface === "current sheet") {
      expect([...view.container.querySelectorAll("td.label")].map(cell => cell.textContent)).toEqual(SHEET_ROWS.map(row => row.label));
    }
    for (let revision = 1; revision <= 4; revision++) {
      const note = `Reviewer revision ${revision}`;
      canonical.elements[`${rowIds[0]}__note`].value = note;
      canonical.elements[`${rowIds[0]}__note`].version = revision;
      view.rerender(createElement(ArtifactPanel, props));
      const button = screen.getByTestId("artifact-export-xlsx");
      fireEvent.click(button);
      fireEvent.click(button); // Ordinary burst against the disabled current action starts once.
      await waitFor(() => expect(exportFixture.bytes).toHaveLength(revision));
      await waitFor(() => expect((button as HTMLButtonElement).disabled).toBe(false));
      const reopened = new ExcelJS.Workbook();
      await reopened.xlsx.load(exportFixture.bytes[revision - 1]);
      const sheet = reopened.getWorksheet("Q3 variance")!;
      expect(SHEET_ROWS.map((_, index) => sheet.getCell(index + 2, 1).value)).toEqual(SHEET_ROWS.map(row => row.label));
      expect(sheet.getCell("B2").value).toBe(SHEET_ROWS[0].q2);
      expect(sheet.getCell("C2").value).toBe(SHEET_ROWS[0].q3);
      expect(sheet.getCell("D2").formula).toBe("C2/B2-1");
      expect(sheet.getCell("D2").result).toBe(0.24);
      expect(sheet.getCell("D2").numFmt).toBe("0.0%");
      expect(sheet.getCell("E2").value).toBe(note);
      if (surface === "sibling sheet") {
        expect(reopened.getWorksheet("Summary")!.getCell("A1").value).toBe("Review workbook");
        expect(reopened.getWorksheet("Summary")!.getCell("B1").formula).toBe("1+1");
        expect(reopened.getWorksheet("Summary")!.getCell("B1").result).toBe(2);
        expect(reopened.getWorksheet("Summary")!.getCell("B1").numFmt).toBe("0.00");
      }
    }
    const first = new ExcelJS.Workbook();
    await first.xlsx.load(exportFixture.bytes[0]);
    expect(first.getWorksheet("Q3 variance")!.getCell("E2").value).toBe("Reviewer revision 1");
  });
});
